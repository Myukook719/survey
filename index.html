    const urls = [

      "URL3", // https://docs.google.com/forms/d/e/1FAIpQLSeBSG1e9iNZCV2FvKs5KxgBBsxhWMoT0d6aLODRHk6mJ6lfkA/viewform?usp=publish-editor

      "URL4", // https://docs.google.com/forms/d/e/1FAIpQLSd2atfcKAYXmia2QdVwpF7Su__HucMsod8Wl6GDgJsKOQ_V2g/viewform?usp=publish-editor

      "URL5", // https://docs.google.com/forms/d/e/1FAIpQLScxbMFBnleDXK-pwoVniu0Kz8DTVLthfq9gZ8Mn3CMhIoJdEQ/viewform?usp=publish-editor

      "URL6"  // https://docs.google.com/forms/d/e/1FAIpQLSeAZxSzeUXLAezS0ucB1B-6uomvYaXd2UhMK4q2oWQD9JOgbQ/viewform?usp=dialog

    ];


    // この調査専用の保存名です。回収中は変更しないでください。

    const storageKey = "student_survey_b_assignment_v1";


    function redirectToForm() {

      // URLの置き換え忘れ・形式の誤りを確認します。

      const validUrls = urls.every(function (value) {

        try {

          return new URL(value).protocol === "https:";

        } catch (error) {

          return false;

        }

      });


      if (!validUrls) {

        document.getElementById("message").textContent =

          "フォームのURLが正しく設定されていません。調査実施者にお知らせください。";

        return;

      }


      let savedValue = null;


      // 前回の割り当てを読み込みます。

      try {

        savedValue = localStorage.getItem(storageKey);

      } catch (error) {

        // 保存機能が使えない場合も、振り分けは実行します。

      }


      let formNumber = Number(savedValue);


      // 有効な保存番号がなければ、0〜5を同じ確率で選びます。

      if (

        savedValue === null ||

        savedValue.trim() === "" ||

        !Number.isInteger(formNumber) ||

        formNumber < 0 ||

        formNumber >= urls.length

      ) {

        formNumber = Math.floor(Math.random() * urls.length);


        try {

          localStorage.setItem(storageKey, String(formNumber));

        } catch (error) {

          // 保存できない場合は、開き直すと再抽選になります。

        }

      }


      // 選ばれたフォームへ移動します。

      window.location.replace(urls[formNumber]);

    }


    redirectToForm();

  </script>
